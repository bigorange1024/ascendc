\begin{frame}{定义：依赖闭包 $\mathrm{Dep}^*$}
\begin{definition}[单点依赖闭包（教材 Def.）]
固定依赖图 $G=(V,E)$。对目标 $T\in V$，
\[
\mathrm{Dep}^*_G(T)
\;=\;
\{u\in V\mid u\rightsquigarrow_G T\}
\;=\;
\{u\in V\mid (u,T)\in E^{*}\}.
\]
恒有 $T\in\mathrm{Dep}^*_G(T)$。工程读法：完成 $T$ 理论上必须齐备的全部先决（含 $T$）。
\end{definition}
\begin{theorem}[迭代收敛]
$D_0(T)=\{T\}$，$D_{k+1}=D_k\cup\bigcup_{v\in D_k}\mathrm{Pred}(v)$。
有限图上存在 $K\le|V|$ 使 $k\ge K\Rightarrow D_k(T)=\mathrm{Dep}^*(T)$。
\end{theorem}
\vspace{-0.3em}
{\footnotesize 等价算法：在转置图 $G^\top$ 上从 $T$ 做 BFS/DFS，访问集 $=\mathrm{Dep}^*(T)$。}
\end{frame}

% ========== 3 ==========
\begin{frame}{定义：证书、$\Gamma$、缺项}
\begin{definition}[证书 / 已获证集 $\Gamma$]
证书 $=$ 对顶点 $v$ 的\textbf{可引用判决}（约定验收下已通过）。\\
$\Gamma\subseteq V$：当前允许被后续任务引用的能力集。$v\in\Gamma$ 称已获证。
\end{definition}
\begin{definition}[缺项集]
\[
\mathrm{Missing}_\Gamma(T)
\;=\;
\mathrm{Dep}^*(T)\setminus\Gamma.
\]
$\mathrm{Missing}_\Gamma(T)=\emptyset\iff\mathrm{Dep}^*(T)\subseteq\Gamma$（闭包已满）。
\end{definition}
\begin{theorem}[缺项基本性质（摘）]
若 $\Gamma\subseteq\Gamma'$ 则 $\mathrm{Missing}_{\Gamma'}(T)\subseteq\mathrm{Missing}_\Gamma(T)$；
且 $\mathrm{Dep}^*(T)=(\mathrm{Dep}^*(T)\cap\Gamma)\,\dot\cup\,\mathrm{Missing}_\Gamma(T)$。
\end{theorem}
\end{frame}

% ========== 4 ==========
\begin{frame}{定义：闭包表 $\mathrm{CT}$ 与三类边}
\begin{definition}[闭包表]
任务启动时填写的检查表，至少含
$(A,\Gamma,\mathrm{Dep}^*(A),\mathrm{Missing}_\Gamma(A))$；
工程扩展栏：接缝 $\mathrm{Seams}$、否定 $\mathrm{ForbiddenHit}$、契约 $\mathcal{C}$、级别 $\ell$。\\
\textbf{先交闭包表，再写码}——customspec/STATUS \textbf{替代不了} $\mathrm{CT}$。
\end{definition}
{\footnotesize
\begin{center}
\begin{tabular}{@{}L{2.2cm}L{4.0cm}L{5.5cm}@{}}
  \toprule
  标签 $\tau$ & 含义 & 写入 / 后果 \\
  \midrule
  $\mathsf{semantic}$ & $v$ 依赖 $u$ 的 interface & 入 $E$；进 $\mathrm{Dep}^*$ \\
  $\mathsf{compose}$ & $A,B$ 各有证，$A\!\circ\!B$ 仍须单证 & 接缝顶点 $s$；\textbf{不}自动入 $\Gamma$ \\
  $\mathsf{forbidden}$ & 禁止路线/模板 & 入 $\mathrm{Forbidden}$ \\
  \bottomrule
\end{tabular}
\end{center}
}
\begin{theorem}[组合不免费]
$A,B\in\Gamma$ 且 $\mathrm{compose}(A,B)$ 得接缝 $s$ $\;\not\Rightarrow\;$ $s\in\Gamma$。
\end{theorem}
\end{frame}

% ========== 5 ==========
\begin{frame}{定理：写码许可 $\mathrm{PermitCode}$（v0）}
\begin{theorem}[写码许可的充分必要条件（教材 Thm.）]
在 $\mathrm{BuildClosureTable}$ 输出的 $\mathrm{CT}$ 上定义
\[
\mathrm{PermitCode}(\mathrm{CT})
\;:\iff\;
M=\emptyset
\;\land\;
\mathrm{Seams}\subseteq\Gamma
\;\land\;
\mathrm{ForbiddenHit}=\emptyset
\;\land\;
\mathcal{C}\text{ 已声明}.
\]
则：（1）为真时可按 $\ell$ 实现 $T$；
（2）$M\neq\emptyset$ 或接缝未获证时，跳过 $\mathsf{Probe}/\mathsf{Certify}$ 写顶层 $=$ 非法转移；
（3）$\mathrm{ForbiddenHit}\neq\emptyset$ 属拒绝类行为。
\end{theorem}
\vspace{0.3em}
{\footnotesize
$\mathrm{GateCheck}$：无表 / Forbidden 相交 / 缺项或接缝未齐 $\to\mathsf{BLOCK}(R)$；否则 $\mathsf{ALLOW}$。\\
外层 $\mathrm{TaskWorkflow}$：声明 $\to$ 展开 $\mathrm{CT}$ $\to$ 补洞直至 $\mathsf{ALLOW}$ $\to$ $\mathsf{Certify}$；变更则 $\mathrm{DirtyPropagate}$。
}
\end{frame}

  2 & 闭包展开 &
  $\mathrm{BuildClosureTable}$；标 ok/missing/dirty；接缝与否定栏齐 \\
  3 & 最小补洞 &
  对 $\mathsf{BLOCK}$ 中缺项跑 $\mathrm{MinimalFillRound}$；forbid 落笔 \\
  4 & 闭环写回 &
  $\mathsf{Certify}(T)$ 入 INDEX/STATUS/节点卡；该脏则 $\mathrm{DirtyPropagate}$ \\
  \bottomrule
\end{tabular}
\end{center}
\vspace{0.5em}
\begin{alertblock}{与转移的对照}
闭包展开 $\leftrightarrow\mathsf{Expand}$；补洞 $\leftrightarrow\mathsf{Probe}+\mathsf{Certify}$；
命中 Forbidden $\leftrightarrow$ 拒绝/冻结；接口变更 $\leftrightarrow\mathsf{Dirty}$。
\end{alertblock}
\end{frame}
