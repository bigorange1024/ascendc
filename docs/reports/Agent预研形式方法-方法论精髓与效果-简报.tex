\begin{frame}{主题：依赖闭包 $\mathrm{Dep}^*$——「完成目标理论上要齐谁」}
\begin{definition}[单点依赖闭包（教材）]
固定 $G=(V,E)$，$T\in V$，
\[
\mathrm{Dep}^*_G(T)
=\{u\in V\mid u\rightsquigarrow_G T\}
=\{u\in V\mid (u,T)\in E^{*}\}.
\]
恒有 $T\in\mathrm{Dep}^*_G(T)$。
\end{definition}
\begin{theorem}[迭代收敛]
$D_0=\{T\}$，$D_{k+1}=D_k\cup\bigcup_{v\in D_k}\mathrm{Pred}(v)$；
有限图上 $\exists K\le|V|$ 使 $k\ge K\Rightarrow D_k=\mathrm{Dep}^*(T)$。
\end{theorem}
{\footnotesize 算法：在 $G^\top$ 上从 $T$ 做 BFS/DFS，访问集 $=\mathrm{Dep}^*(T)$。}
\pagesummary{给出「先决集合」的精确定义，使「缺什么」可计算，而不是凭感觉列清单。}
\end{frame}

% ========== 3 ==========
\begin{frame}{主题：证书 $\Gamma$ 与缺项 $\mathrm{Missing}$——「已有什么、还缺什么」}
\begin{definition}[证书 / $\Gamma$]
证书 $=$ 对顶点 $v$ 的\textbf{可引用判决}。$\Gamma\subseteq V$：当前允许引用的能力集。
\end{definition}
\begin{definition}[缺项]
$\mathrm{Missing}_\Gamma(T)=\mathrm{Dep}^*(T)\setminus\Gamma$；
空集 $\iff$ 闭包已满（$\mathrm{Dep}^*(T)\subseteq\Gamma$）。
\end{definition}
\begin{theorem}[缺项基本性质（摘）]
$\Gamma\subseteq\Gamma'\Rightarrow\mathrm{Missing}_{\Gamma'}(T)\subseteq\mathrm{Missing}_\Gamma(T)$；
$\mathrm{Dep}^*(T)=(\mathrm{Dep}^*(T)\cap\Gamma)\,\dot\cup\,\mathrm{Missing}_\Gamma(T)$。
\end{theorem}
\pagesummary{把「能不能引用」与「还要补什么」从口语变成集合差，后面门禁直接读这两个对象。}
\end{frame}

% ========== 4 ==========
\begin{frame}{主题：闭包表 $\mathrm{CT}$ 与三类边——写码前交什么、边怎么分类}
\begin{definition}[闭包表]
至少含 $(A,\Gamma,\mathrm{Dep}^*(A),\mathrm{Missing}_\Gamma(A))$；
扩展：$\mathrm{Seams}$、$\mathrm{ForbiddenHit}$、$\mathcal{C}$、$\ell$。\\
\textbf{先交 $\mathrm{CT}$，再写码}——customspec/STATUS \textbf{替代不了} $\mathrm{CT}$。
\end{definition}
{\footnotesize
\begin{center}
\begin{tabular}{@{}L{2.1cm}L{3.8cm}L{5.4cm}@{}}
  \toprule
  $\tau$ & 含义 & 后果 \\
  \midrule
  $\mathsf{semantic}$ & $v$ 依赖 $u$ 的 interface & 入 $E$，进 $\mathrm{Dep}^*$ \\
  $\mathsf{compose}$ & $A,B$ 各有证，$A\!\circ\!B$ 仍须单证 & 接缝 $s$\textbf{不}自动入 $\Gamma$ \\
  $\mathsf{forbidden}$ & 禁止路线/模板 & 入 $\mathrm{Forbidden}$ \\
  \bottomrule
\end{tabular}
\end{center}
}
\begin{theorem}[组合不免费]
$A,B\in\Gamma$ 且得接缝 $s$ $\;\not\Rightarrow\;$ $s\in\Gamma$。
\end{theorem}
\pagesummary{规定写码前的审计表单，并堵住「两块已绿就默认拼装已绿」的幻觉。}
\end{frame}

% ========== 5 ==========
\begin{frame}{主题：写码许可定理 $\mathrm{PermitCode}$——何时允许实现目标 $T$}
\begin{theorem}[写码许可的充要条件（v0，教材）]
\[
\mathrm{PermitCode}(\mathrm{CT})
:\iff
M=\emptyset
\land\mathrm{Seams}\subseteq\Gamma
\land\mathrm{ForbiddenHit}=\emptyset
\land\mathcal{C}\text{ 已声明}.
\]
为真 $\Rightarrow$ 可按 $\ell$ 实现；缺项/接缝未齐或踩 Forbidden $\Rightarrow$ 非法转移，应 $\mathsf{BLOCK}$。
\end{theorem}
{\footnotesize
$\mathrm{GateCheck}\to\mathsf{ALLOW}/\mathsf{BLOCK}(R)$；
$\mathrm{TaskWorkflow}$：声明$\to\mathrm{CT}\to$补洞$\to\mathsf{Certify}$；变更则 $\mathrm{DirtyPropagate}$。
}
\pagesummary{把门禁从口号收成可判定谓词，Agent/人能否开写有同一把尺子。}
\end{frame}
